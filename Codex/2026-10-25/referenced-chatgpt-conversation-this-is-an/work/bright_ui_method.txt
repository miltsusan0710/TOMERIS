    def change_text_size(self,delta):
        self.chat_size=max(10,min(18,self.chat_size+delta))
        self.transcript.configure(font=('Yu Gothic UI',self.chat_size))
        self.entry.configure(font=('Yu Gothic UI',self.chat_size))
        for tag in ('you','assistant'):
            self.transcript.tag_configure(tag,font=('Yu Gothic UI',self.chat_size,'bold'))
        self.transcript.tag_configure('notice',font=('Yu Gothic UI',max(10,self.chat_size-1)))
